# v2.1 — Special research project decisions (requires Gotterdammerung).
# Each effect instantly completes one specialization tree via the vanilla
# complete_special_project effect, so every project's own project_output
# (technologies, equipment unlocks, module unlocks, free nuclear bombs)
# fires exactly as if the project had been researched normally.
#
# Projects are listed parent-first because special_project_parent chains
# span the tree. Cross-tree parents exist (naval needs sp_nuclear_engines
# and sp_rockets_ballistic_missile; nuclear warheads need the MRBMs; the
# LRBM needs nuclear warheads), so the decisions are re-runnable: if a
# project is skipped because a cross-tree parent is not completed yet,
# running the other groups and clicking again finishes the rest.
#
# Each effect also grants that tree's derived technologies — vanilla techs
# flagged is_special_project_tech = yes whose allow block requires the
# completed project — plus the hidden "pick" techs normally granted by
# mutually exclusive prototype-reward choices (naval support/repair/
# replenishment variants, centrifugal isotope separation). Instant
# completion never fires iteration rewards, so all choices are granted.
#
# Script completion has no facility state, so facility_state_effects are
# skipped by the engine. The only loss worth compensating is the free
# level-1 nuclear reactor from sp_nuclear_reactor, granted in the capital.
# Prototype iteration rewards (scientist XP, progress boosts, setback
# events) are tied to running real iterations and never fire here.

PRC_OCS_complete_air_special_projects_effect = {

			complete_special_project = sp:sp_air_bouncing_bomb

		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_air_earthshaker_bomb } }
			complete_special_project = sp:sp_air_earthshaker_bomb
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_air_intercontinental_bomber } }
			complete_special_project = sp:sp_air_intercontinental_bomber
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_air_mothership_aircraft } }
			complete_special_project = sp:sp_air_mothership_aircraft
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_air_jet_engine } }
			complete_special_project = sp:sp_air_jet_engine
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_air_axial_jet_engine } }
			complete_special_project = sp:sp_air_axial_jet_engine
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_air_supersonic_jet } }
			complete_special_project = sp:sp_air_supersonic_jet
		}
		if = {

			complete_special_project = sp:sp_air_helicopter
		}

		if = {

PRC_OCS_complete_nuclear_special_projects_effect = {
	if = {
		limit = { has_dlc = "Gotterdammerung" }
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_nuclear_reactor } }
			complete_special_project = sp:sp_nuclear_reactor
			# Script completion skips the free level-1 reactor normally built
			# in the facility state; grant it in the capital instead.
			capital_scope = {
				add_building_construction = {
					type = nuclear_reactor
					level = 1
					instant_build = yes
				}
			}
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_commercial_nuclear_reactor } }
			complete_special_project = sp:sp_commercial_nuclear_reactor
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_nuclear_engines } }
			complete_special_project = sp:sp_nuclear_engines
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_nuclear_bomb } }
			complete_special_project = sp:sp_nuclear_bomb
		}
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_thermo_nuclear_bomb } }
			complete_special_project = sp:sp_thermo_nuclear_bomb
		}
		# Parents: nuclear bomb + sp_rockets_medium_range_ballistic_missile.
		if = {
			limit = { NOT = { is_special_project_completed = sp:sp_nuclear_warheads } }
			complete_special_project = sp:sp_nuclear_warheads
		}
		# Prototype-reward choice tech: the centrifugal option of the isotope
		# separation reward (+50% nuclear production). Never fires here.
		set_technology = {
			isotope_separation_centrifugal = 1
		}
		set_country_flag = PRC_OCS_nuclear_special_projects_completed
		country_event = { id = PRC_OCS.12 }
	}
}
