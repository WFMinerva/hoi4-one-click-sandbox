# v2.0 — Orchestration layer
# This file retains the three orchestration effects (resupply, add_points,
# initialize) that compose the domain-specific effects defined in the
# companion files under common/scripted_effects/.
#
# Domain effects have been split into:
#   PRC_OCS_research_effects.txt      — research_all, configure_doctrines, configure_spirits
#   PRC_OCS_mio_effects.txt           — boost_all_mios, appoint_advisors, configure_mios
#   PRC_OCS_equipment_effects.txt     — create_variants, create_generic_variants
#   PRC_OCS_template_effects.txt      — create_templates, create_generic_templates
#   PRC_OCS_stockpile_effects.txt     — add_stockpile, add_generic_stockpile
#   PRC_OCS_construction_effects.txt  — unlock_all_shared_slots, queue_transport_and_bases,
#                                        queue_petrochemical_program, queue_coastal_dockyards,
#                                        queue_civilian_industry, queue_national_defenses,
#                                        unlock_current_state_shared_slots,
#                                        refresh_industrial_priority_flags,
#                                        build_target_supply_nodes
#   PRC_OCS_military_effects.txt      — raise_24_infantry_divisions, reorganize_all_divisions
#   PRC_OCS_special_project_effects.txt — complete air/land/naval/nuclear/rocket
#                                        special project trees (Gotterdammerung)
#
# The engine loads every .txt file under common/scripted_effects/ automatically,
# so resolution of effect names across files works without any extra wiring.

PRC_OCS_resupply_effect = {
	if = {
		limit = {
			OR = {
				tag = PRC
				original_tag = PRC
			}
		}
		PRC_OCS_add_stockpile_effect = yes
	}
	else = {
		PRC_OCS_add_generic_stockpile_effect = yes
	}
}

PRC_OCS_add_points_effect = {
	add_political_power = 99999
	add_command_power = 100
	army_experience = 999
	navy_experience = 999
	air_experience = 999
}

# v2.0b: generic countries now receive neutral equipment designs, eight division
# templates and the full supported stockpile. PRC retains its Chinese design names,
# advisers and configured MIO traits. Existing divisions are never changed here.
PRC_OCS_initialize_effect = {
	PRC_OCS_research_all_effect = yes
	PRC_OCS_configure_doctrines_effect = yes
	PRC_OCS_configure_spirits_effect = yes
	PRC_OCS_configure_shared_mios_effect = yes

	if = {
		limit = {
			OR = {
				tag = PRC
				original_tag = PRC
			}
		}
		PRC_OCS_appoint_advisors_effect = yes
		PRC_OCS_configure_mios_effect = yes
		PRC_OCS_create_variants_effect = yes
		PRC_OCS_create_templates_effect = yes
		PRC_OCS_add_stockpile_effect = yes
		set_country_flag = PRC_OCS_initialized
		country_event = { id = PRC_OCS.1 }
	}
	else = {
		PRC_OCS_create_generic_variants_effect = yes
		PRC_OCS_create_generic_templates_effect = yes
		PRC_OCS_add_generic_stockpile_effect = yes
		set_country_flag = PRC_OCS_initialized
		country_event = { id = PRC_OCS.7 }
	}
}
