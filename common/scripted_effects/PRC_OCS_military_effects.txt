# v2.2-test1 universal repeatable force-raising decision. The template name resolves
# to the PRC militia-based version for PRC/original-PRC countries and to the
# generic infantry-based version for every other country.
PRC_OCS_raise_24_infantry_divisions_effect = {
	random_owned_controlled_state = {
		create_unit = {
			division = "division_template = \"主力纯步师\" start_experience_factor = 1.0 start_equipment_factor = 1.0 start_manpower_factor = 1.0"
			owner = ROOT
			allow_spawning_on_enemy_provs = no
			count = 24
		}
	}
	country_event = { id = PRC_OCS.11 hours = 6 }
}

# Universal, one-time and internally idempotent. PRC/original-PRC countries use
# the militia/name-group templates; every other country uses the infantry-based
# generic templates. No existing or active division is rewritten.
PRC_OCS_evolve_special_project_forces_effect = {
	if = {
		limit = {
			OR = {
				tag = PRC
				original_tag = PRC
			}
		}
		PRC_OCS_create_evolved_variants_effect = yes
		PRC_OCS_create_evolved_templates_effect = yes
	}
	else = {
		PRC_OCS_create_evolved_variants_effect = yes
		PRC_OCS_create_generic_evolved_templates_effect = yes
	}
	set_country_flag = PRC_OCS_special_project_forces_evolved
	country_event = { id = PRC_OCS.13 }
}

PRC_OCS_reorganize_all_divisions_effect = {
	every_country_division = {
		change_division_template = {
			division_template = "卫戍部队"
		}
	}
	set_country_flag = PRC_OCS_army_reorganized
	country_event = { id = PRC_OCS.8 }
}
