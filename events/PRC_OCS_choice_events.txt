add_namespace = PRC_OCS

# v2.6 choose-your-bonus event chain (air special projects).
# The one-click completion instantly completes special projects, so the engine
# never fires prototype iteration rewards. These events let the player pick
# each mutually exclusive reward buff one by one. Each buff group is tracked
# by its own country flag; the dispatch event walks the unpicked groups in
# order and marks the chain done when all are picked.

country_event = {
	id = PRC_OCS.14
	title = PRC_OCS.14.t
	desc = PRC_OCS.14.d
	picture = GFX_report_event_generic_research
	is_triggered_only = yes

	option = {
		name = PRC_OCS.14.a
		trigger = { NOT = { has_country_flag = PRC_OCS_air_jet_choice_done } }
		country_event = { id = PRC_OCS.15 }
	}
	option = {
		name = PRC_OCS.14.b
		trigger = {
			has_country_flag = PRC_OCS_air_jet_choice_done
			NOT = { has_country_flag = PRC_OCS_air_axial_choice_done }
		}
		country_event = { id = PRC_OCS.16 }
	}
	option = {
		name = PRC_OCS.14.c
		trigger = {
			has_country_flag = PRC_OCS_air_axial_choice_done
			NOT = { has_country_flag = PRC_OCS_air_supersonic_choice_done }
		}
		country_event = { id = PRC_OCS.17 }
	}
	option = {
		name = PRC_OCS.14.e
		trigger = {
			has_country_flag = PRC_OCS_air_supersonic_choice_done
		}
		country_event = { id = PRC_OCS.50 }
	}
}
